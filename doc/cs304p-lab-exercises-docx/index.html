Lab 03
Problem Statement
Write a program which consists of a class named Room having two data members Height and Width, the class should also consist of three constructors i.e., Default constructor, one argument constructor and two arguments constructor.
Sample Output
After running your program, the following screen should display.

Solution
#include<iostream>
using namespace std;
class Room
{
	private:        //private variables of type float.
	float width;
	float height;
	public:
		Room()  //Default Constructor
		{
			width=0.0;   //Default value has been assigned to width
			height=0.0; //Default value has been assigned to height
			cout<<"Default Constructor is called" <<endl;
			cout<<"Height: "<<width<<endl;
			cout<<"Width: "<<height<<endl<<endl;
		}
		Room(float h)  //parametrized Constructor (One argument)
		{
			height=h;
			width=0.0;
			cout<<"One Argument Constructor is called" <<endl;
			cout<<"Height: "<<width<<endl;
			cout<<"Width: "<<height<<endl<<endl;
		}
		Room(float h, float w)  //parametrized Constructor (Two argument)
		{
			height=h;
			width=w;
			cout<<"Two Argument Constructor is called" <<endl;
			cout<<"Height: "<<width<<endl;
			cout<<"Width: "<<height<<endl<<endl;
		}
};
int main()
{
	Room R1, R2(10), R3(20,30);
	return 0;
}
Lab 04
Problem Statement
Write C++ coding program which creates a class Pyramid. The class should contain a static data member “Pcount” to store the total number of Pyramid’s objects.
In the main() function, dynamically create three objects of Pyramid class and print the value of “Pcount”. Afterwards, delete any two objects of Pyramid and print the value of “Pcount” again.
Sample Output
Following is a sample output for above scenario:

Solution
#include<iostream>
using namespace std;
class Pyramid
{
	private:   //private static variable of type int
		static int Pcount;
	public:
		static int get_pcount();    //static function get_pcount() and its retrun type is int.
		Pyramid();  //Constructor
		~Pyramid();   //Destructor
};
main()
{
//This statement is creating an object dynamically and binds its address to a pointer.
//This is the declaration of variable named p1, p2, and p3 which is a pointer to a Pyramid. This pointer object has automatic storage duration. It is then initialized with the result of new Pyramid(). This new keyword creates an p object with dynamic storage duration and then returns a pointer to it.
Pyramid *p1 = new Pyramid;
	Pyramid *p2 = new Pyramid;
	Pyramid *p3 = new Pyramid;

//prints the number of total objects before deletion.
cout << endl << "Total Objects :  " << Pyramid::get_pcount() << endl << endl;

	delete p1;      //Delete the object p1
	delete p2;     //Delete the object p2

           //prints the number of total objects after deletion.
             cout << endl << "Total Objects :  " << Pyramid::get_pcount() << endl << endl;

	system("pause");
}
int Pyramid::Pcount = 0;

int ::Pyramid::get_pcount()   //definition of function get_pcount

{
	return Pcount;
}
Pyramid::Pyramid()  //definition of constructor
{
	cout << endl << "Pyramid Constructor Called.." << endl;
	Pcount++;
}
Pyramid::~Pyramid()  //definition of destructor
{
	cout << endl << "Pyramid Destructor Called.." << endl;
	Pcount--;
}
