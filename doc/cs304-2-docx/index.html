int TicTacToe::gameStatus()
{
	if (scr[0][0] == scr[0][1] && scr[0][1] == scr[0][2] && scr[0][0] != ' ')

        return 1;
    else if (scr[1][0] == scr[1][1] && scr[1][1] == scr[1][2] && scr[1][0] != ' ')

        return 1;
    else if (scr[2][0] == scr[2][1] && scr[2][1] == scr[2][2] && scr[2][0] != ' ')

        return 1;

    else if (scr[0][0] == scr[1][0] && scr[1][0] == scr[2][0] && scr[0][0] != ' ')

        return 1;
    else if (scr[0][1] == scr[1][1] && scr[1][1] == scr[2][1] && scr[0][1] != ' ')

        return 1;
    else if (scr[0][2] == scr[1][2] && scr[1][2] == scr[2][2] && scr[0][2] != ' ')

        return 1;
    else if (scr[0][0] == scr[1][1] && scr[1][1] == scr[2][2] && scr[0][0] != ' ')

        return 1;
    else if (scr[0][2] == scr[1][1] && scr[1][1] == scr[2][0] && scr[0][2] != ' ')

        return 1;
    else if (scr[0][0] != ' ' && scr[0][1] != ' ' && scr[0][2] != ' '
                    && scr[1][0] != ' ' && scr[1][1] != ' ' && scr[1][2] != ' '

}
